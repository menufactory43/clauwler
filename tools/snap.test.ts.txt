import { expect, test } from 'claude-code/testing'
import { blits, boot, pane } from './pane.test'

// Plays the real mod in the test bench and prints the arena's cells at a few moments.
test('snap', async ($, on) => {
  const clock = await boot($, on, '__TERM__')
  const ui = await $.ui.mount({ plugin: 'clauwler', surface: 'terminal', ...pane(true, Number('__COLS__')) } as never) as any
  const shot = async (name: string) => {
    const arena = await ui.find({ key: 'arena' })
    const p = arena?.props as { columns: number; rows: number; cells: string } | undefined
    if (blits.image?.png) console.log(`PNG ${name} ${blits.image.png}`)
    else if (blits.image) console.log(`IMG ${name} ${blits.image.width} ${blits.image.height} ${blits.image.rgba}`)
    else console.log(`SNAP ${name} ${p ? `${p.columns} ${p.rows} ${blits.cells ?? p.cells}` : 'none'}`)
    console.log(`TEXT ${name} blits=${blits.count}`)
    const texts = (await ui.findAll({ type: 'Text' })).map((t: any) => t.text).join(' | ')
    console.log(`TEXT ${name} ${texts}`)
  }
  const hold = async (key: string, ms: number) => {
    for (let t = 0; t < ms; t += 84) {
      await ui.press({ key: `p-${key}` })
      await clock.advance(84)
    }
  }
  await clock.advance(200)
  if (await ui.find({ key: 'o-0' })) await ui.press({ key: 'o-0' })
  await clock.advance(1500)
  await shot('start')
  // Walk out through the first door found, then fight (the champion strikes by himself).
  for (const key of ['s', 'z', 'd', 'q']) {
    await hold(key, 1600)
    if ((await ui.findAll({ type: 'Text' })).some((t: any) => /Salle des|Couloir|Crypte|Archives|Puits|Galerie|Forge|Chapelle|Erreurs/.test(t.text))) break
  }
  await clock.advance(1200)
  await shot('room')
  await hold('d', 700)
  await hold('r', 300)
  await hold('z', 500)
  await shot('fight')
  await ui.press({ key: 'p-p' })
  await clock.advance(200)
  await shot('pause')
  await ui.press({ key: 'p-h' })
  await clock.advance(200)
  const sp = await ui.find({ key: 'splash' })
  console.log(`SNAP hall ${sp.props.columns} ${sp.props.rows} ${sp.props.cells}`)
  expect(true).toBe(true)
})
